# Clicking a folder in the tags dock opens it in a new tab; clicking it again
# activates that tab instead of opening a duplicate. The context menu's
# "Open in current tab" navigates the focused pane instead.
#   ./scripts/ui-probe.ps1 -Script scripts/probe/tags_click_new_tab.txt -Seed scripts/probe/seed_tags
size 1117 717
wait_ms 1500
# Open the dock focused on "Work".
click 80 294
move 700 300
wait_ms 300
shot out/tags_click_0_dock.png
# Click "Windows" in the Work group: new tab "Windows", active.
click 350 557
move 700 300
wait_ms 600
shot out/tags_click_1_new_tab.png
# Back to the "This PC" tab, click "Windows" again: the existing tab is
# activated (still 2 tabs).
click 320 20
move 700 300
wait_ms 400
shot out/tags_click_2_back_to_pc.png
click 350 557
move 700 300
wait_ms 600
shot out/tags_click_3_activated.png
# Right-click "Users": context menu with "Open in current tab".
click 350 583 right
wait_ms 300
shot out/tags_click_4_context.png
# "Open in current tab": the Windows tab navigates to Users (still 2 tabs).
click 410 601
move 700 300
wait_ms 600
shot out/tags_click_5_current_tab.png
quit
