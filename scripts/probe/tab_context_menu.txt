# Tab right-click menu, vertical rail (run with -Seed scripts/probe/seed_vertical_tabs).
# Launch with a scratch folder containing subfolders alpha/beta/gamma (ui-probe.ps1
# passes no args), e.g. from bash:
#   mkdir -p "$TEMP/eden-tabctx-scratch/"{alpha,beta,gamma}
#   mkdir -p "$TEMP/eden-tabctx-data"; cp scripts/probe/seed_vertical_tabs/* "$TEMP/eden-tabctx-data/"
#   EDEN_PROBE_SCRIPT=<abs path of this file> EDEN_PROBE_DATA_DIR=<that data dir> \
#     target/probe/release/EdenExplorer.exe "%TEMP%\eden-tabctx-scratch"
# Rows: y=100 / 132 / 164 / 196. The menu opens at the pointer; items are 22px apart
# starting 17px below it (Add tag, Favorites, Pin, Duplicate, [Open in split view],
# Copy path, separator, Close, Close others, Close below). "Copy path" is never
# clicked because it writes the real clipboard.
